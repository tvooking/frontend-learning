📦 ES6+ 现代 JavaScript — 箭头函数与解构

📌 前置概念：let 和 const（复习）

在基础阶段学过了，现在记住一条原则：尽量用 const，需要改的用 let，永远不用 var。

const PI = 3.14;       // 常量，不能重新赋值
let count = 0;          // 变量，可以改
// var old = "别用我";  // ❌ 不要用 var

➡️ 箭头函数 Arrow Function

// 传统函数
function add(a, b) { return a + b; }

// 箭头函数（等价）
const add = (a, b) => a + b;

// 如果只有一个参数，可以省略括号
const double = x => x * 2;

// 如果没有参数，必须写括号
const hello = () => "你好！";

// 如果函数体有多行，需要加 {} 和 return
const greet = (name) => {
  const msg = `你好，${name}！`;
  return msg;
};

箭头函数和传统函数的区别：
✅ 语法更简洁
✅ 不绑定自己的 this（后面会用到这个优势）

🔄 模板字符串 Template Literals

const name = "小明";
const age = 25;

// 旧方式：字符串拼接
const old = "我叫" + name + "，今年" + age + "岁。";

// 新方式：模板字符串（反引号 + ${}）
const msg = `我叫${name}，今年${age}岁。`;

// 还支持多行
const html = `
  <div>
    <p>${name}</p>
  </div>
`;

🔀 解构赋值 Destructuring

// ===== 数组解构 =====
const colors = ["red", "green", "blue"];
const [first, second, third] = colors;
console.log(first);   // "red"

// 跳过元素
const [a, , c] = colors;
console.log(c);        // "blue"

// ===== 对象解构 =====
const person = { name: "小明", age: 25, city: "北京" };
const { name, age, city } = person;
console.log(name);     // "小明"

// 重命名
const { name: userName, age: userAge } = person;
console.log(userName); // "小明"

// 默认值
const { hobby = "无" } = person;
console.log(hobby);    // "无"

// 嵌套解构
const data = { user: { name: "小红", age: 22 } };
const { user: { name: n } } = data;
console.log(n);        // "小红"

🌊 展开运算符 Spread Operator

// 数组展开
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];
const merged = [...arr1, ...arr2];  // [1,2,3,4,5,6]

// 复制数组（浅拷贝）
const copy = [...arr1];

// 对象展开
const base = { name: "小明", age: 25 };
const withCity = { ...base, city: "北京" };
// { name: "小明", age: 25, city: "北京" }

// 覆盖属性
const updated = { ...base, age: 26 };
// { name: "小明", age: 26 }
